-BonusData：每圈 Bonus 得分（下标对应重转序号）
-BonusPos：
每圈一个数组。第1个数是位置掩码，后面是香蕉金额（从左到右）
掩码 bit0~14=香蕉所在格，bit16~30=猩猩所在格。
例：[42082324, 725, 500, 375] 
低 16 位 = 香蕉图 = 42082324 & 65535 = 8212   =>8212 = 2^2 + 2^4 + 2^13  =>香蕉在 2、4、13格
高 16 位 = 猩猩图 = 42082324 >> 16  = 642       => 642=2^1 +2^7 +2^9 	=>猩猩在  1、7、9格
还原：[0, 20001, 30725, 0, 30500, 0, 0, 20001, 0, 20001, 0, 0, 0, 30375, 0]


-BonusBet：BonusData 求和

/void GameResult_3994_OutResToSenv(OutResult_t* outRes, GameInstance_t* inst, int32_t* res, GameInstanceId_t gameId)
{
	int32_t pos = 0;
	uint8_t wheelChessNum = inst->gameConfig.header.wheelChessNum;

	res[pos++] = (int32_t)gameId;						// ret
	res[pos++] = (int32_t)outRes->openType;				// OpenType
	res[pos++] = (int32_t)outRes->resType;				// ResultType
	res[pos++] = outRes->matrix.idVecSize;				// lineNum
	res[pos++] = outRes->nMatrixBet;					// TotalBet
	res[pos++] = wheelChessNum;							// MatrixLength
	// IDVec
	for (int32_t i = 0; i < outRes->matrix.idVecSize; i++)
	{
		res[pos++] = outRes->IDVec[i];
	}
	//Matrix
	for (int32_t i = 0; i < wheelChessNum; i++)
	{
		res[pos++] = outRes->matrix.dataArray[i];
	}

	if (outRes->resType == RT_FreeWin)
	{
		res[pos++] = outRes->nTotalFreeTime;					// nTotalFreeTime
		res[pos++] = outRes->nTotalFreeBet;					// nTotalFreeBet
		//FreeBetArray
		for (int32_t i = 0; i < outRes->nTotalFreeTime; i++)
		{
			res[pos++] = outRes->FreeBetArray[i];
		}
	}

	if (outRes->openType == OT_Give)
	{

	}

	if (outRes->resType == RT_BonusWin)
	{
		res[pos++] = outRes->BlindSymbol;					//大奖旋转次数
		res[pos++] = outRes->nBonusBet;						// nBonusBet
		for (int32_t i = 0; i < wheelChessNum; i++)
		{
			res[pos++] = outRes->BonusData[i];
		}
		pos = GameResult_3994_WriteBonusPosSenv(outRes, wheelChessNum, res, pos);
	}

	if (outRes->resType == RT_Jackpot)
	{
		res[pos++] = outRes->BlindSymbol;					//大奖旋转次数
		res[pos++] = outRes->nBonusBet;						// nBonusBet
		for (int32_t i = 0; i < wheelChessNum; i++)
		{
			res[pos++] = outRes->BonusData[i];
		}
		pos = GameResult_3994_WriteBonusPosSenv(outRes, wheelChessNum, res, pos);

		res[pos++] = outRes->nJPCount;
		for (int32_t i = 0; i < outRes->nJPCount; ++i)
		{
			res[pos++] = outRes->JPTypeArray[i];
		}
		for (int32_t i = 0; i < outRes->nJPCount; ++i)
		{
			res[pos++] = outRes->JPBetArray[i];
		}
		res[pos++] = outRes->nTotalJackpotBet;
	}
}


//c#解析BonusPos参考代码
public static class GameResult3994SenvParser
{
    public const int ResultTypeBonus = 3;
    public const int ResultTypeJackpot = 4;
    public const int GorillaValue = 20001;
    public const int BananaBase = 30000;

    /// <summary>
    /// 从 senv res[] 解析每圈 15 格盘面。
    /// 格值：0=空，20001=猩猩，30000+金额=香蕉。
    /// </summary>
    public static List<int[]> ParseBonusPosFromRes(int[] res)
    {
        var boards = new List<int[]>();
        if (res == null || res.Length < 6)
            return boards;

        int resultType = res[2];
        if (resultType != ResultTypeBonus && resultType != ResultTypeJackpot)
            return boards;

        int lineNum = res[3];
        int matrixLen = res[5];
        int i = 6 + lineNum + matrixLen;
        // ResultType==2 才会先写免费字段，大奖/彩金不会走到

        if (i + 2 + matrixLen > res.Length)
            return boards;

        int spinCount = res[i++];   // BlindSymbol / TotalBonusSpinTime
        i++;                        // BonusBet
        i += matrixLen;             // BonusData 固定 matrixLen 个

        for (int s = 0; s < spinCount; s++)
        {
            if (i >= res.Length)
                break;

            uint mask = (uint)res[i];
            int nBanana = CountBanana(mask, matrixLen);
            int len = 1 + nBanana;
            if (i + len > res.Length)
                break;

            boards.Add(DecodeSpin(res, i, len, matrixLen));
            i += len;
        }
        return boards;
    }

    static int CountBanana(uint mask, int cellCount)
    {
        uint bananaMap = mask & 65535u;
        int n = 0;
        for (int c = 0; c < cellCount; c++)
        {
            if ((bananaMap & (1u << c)) != 0)
                n++;
        }
        return n;
    }

    static int[] DecodeSpin(int[] res, int start, int len, int cellCount)
    {
        var board = new int[cellCount];
        uint mask = (uint)res[start];
        uint bananaMap = mask & 65535u;
        uint gorillaMap = mask >> 16;

        int p = start + 1;
        int end = start + len;
        for (int c = 0; c < cellCount; c++)
        {
            if ((gorillaMap & (1u << c)) != 0)
                board[c] = GorillaValue;
            else if ((bananaMap & (1u << c)) != 0)
                board[c] = BananaBase + (p < end ? res[p++] : 0);
        }
        return board;
    }
}
